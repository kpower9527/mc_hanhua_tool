## 调研结论（2026-11-21）

- **CFPA 1.12.2**：`autobuild` 发布中的 `Minecraft-Mod-Language-Modpack.zip` + `1.12.2.md5`；`pack_format` 为 3；语言文件为 `assets/<ns>/lang/zh_cn.lang`（1602 个）；另含 1263 个 Patchouli 手册文件（说明 1.12 手册可通过资源包覆盖）
- **I18nUpdateMod**：MC < 1.13 时 options.txt 中的资源包 ID 不带 `file/` 前缀
- **`.lang` 格式**：每行 `key=value`（第一个 `=` 分隔），`#` 开头为注释；Forge 支持首行 `#PARSE_ESCAPES` 开启 `\n` 等转义
- **pack_format**：1.6–1.8 为 1，1.9–1.10 为 2，1.11–1.12 为 3，1.13–1.14 为 4，1.15–1.16.1 为 5
- **RLCraft 2.9.3**：179 个 mod，`.lang` 语言文件为主；无任务书 mod；2 本 Patchouli 手册；7 个 CraftTweaker 脚本
- **Better Questing**：任务存放在 `config/betterquesting/DefaultQuests.json`（单文件）或 `config/betterquesting/DefaultQuests/` 目录（新版拆分格式）。键名带 NBT 类型后缀，文本字段为 `name:8`、`desc:8`。DefaultQuests 只在新建存档或执行 `/bq_admin default load` 时载入
- **FTB Quests 1.12**：任务存放在 `config/ftbquests/normal/` 下的多级 `.snbt` 文件中，文本字段同为 `title`、`description` 等

## Global Constraints

- 前面所有计划的全局约束继续有效
- MC < 1.13 时：资源包 ID 不带 `file/` 前缀；options.txt 中 `resourcePacks` 缺失时默认列表为空（不加 `vanilla`）
- `.lang` 原文按原样保存（不做转义解码）；写出 `zh_cn.lang` 时，若任一原文含反斜杠，则在首行写 `#PARSE_ESCAPES`
- 同一命名空间同时存在 `en_us.json` 与 `en_us.lang` 时，以 `.json` 为准
- Better Questing 条目：`handler="betterquesting"`、`namespace="betterquesting"`、`output="instance"`

```python
def parse_lang_legacy(data: bytes) -> dict[str, str]:
    """1.12 及更早的 .lang 格式：key=value，# 开头为注释，值按原样保留。"""
    out: dict[str, str] = {}
    for line in data.decode("utf-8-sig", errors="replace").splitlines():
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.split("=", 1)
        if key.strip():
            out[key.strip()] = value
    return out


def dump_lang_legacy(mapping: dict[str, str]) -> bytes:
    lines = ["#PARSE_ESCAPES"] if any("\\" in v for v in mapping.values()) else []
    lines += [f"{k}={v}" for k, v in mapping.items()]
    return ("\n".join(lines) + "\n").encode("utf-8")


def parse_any(data: bytes, ext: str) -> dict[str, str]:
    return parse_lang_legacy(data) if ext.lower() == "lang" else parse_lang(data)
```

```python
from __future__ import annotations

import json
import re
from pathlib import Path
from typing import Iterator

from ..applied import read_original
from ..instance import Instance
from ..models import Entry, TranslatedItem
from .base import Handler, OutputSink

BQ_DIR = "config/betterquesting"
TEXT_KEYS = frozenset({"name:8", "desc:8"})
KEY_LIKE_RE = re.compile(r"^[a-z0-9_:-]+(?:\.[a-z0-9_:-]+)+$")


def _walk(node, path: list[str]) -> Iterator[tuple[list[str], str]]:
    if isinstance(node, dict):
        for key, value in node.items():
            if key in TEXT_KEYS and isinstance(value, str):
                yield path + [key], value
            else:
                yield from _walk(value, path + [key])
    elif isinstance(node, list):
        for i, value in enumerate(node):
            yield from _walk(value, path + [str(i)])


def _set(node, path: list[str], value: str) -> None:
    for key in path[:-1]:
        node = node[int(key)] if isinstance(node, list) else node[key]
    if isinstance(node, list):
        node[int(path[-1])] = value
    else:
        node[path[-1]] = value


class BetterQuestingHandler(Handler):
    name = "betterquesting"
    kind_label = "Better Questing 任务书（任务名称与描述）"
    output = "instance"

    def files(self, root: Path) -> list[str]:
        base = Path(root) / BQ_DIR
        found = [base / "DefaultQuests.json"] + sorted((base / "DefaultQuests").rglob("*.json"))
        return [p.relative_to(root).as_posix() for p in found if p.is_file()]

    def detect(self, instance: Instance) -> bool:
        return bool(self.files(instance.root))

    def _load(self, root: Path, rel: str):
        data = read_original(root, rel)
        if data is None:
            return None
        try:
            return json.loads(data.decode("utf-8-sig"), strict=False)
        except ValueError as exc:
            self.warnings.append(f"任务文件解析失败，已跳过：{rel}（{exc}）")
            return None

    def extract(self, instance: Instance) -> Iterator[Entry]:
        self.warnings = []
        for rel in self.files(instance.root):
            obj = self._load(instance.root, rel)
            if obj is None:
                continue
            name = rel.rsplit("/", 1)[-1]
            for path, text in _walk(obj, []):
                if not text.strip() or KEY_LIKE_RE.match(text.strip()):
                    continue
                yield Entry(handler=self.name, source=rel, locator=json.dumps(path, ensure_ascii=False),
                            namespace="betterquesting", original=text, context=f"{name}#{path[-1]}")

    def write(self, items: list[TranslatedItem], sink: OutputSink) -> None:
        if sink.root is None:
            raise ValueError("Better Questing 处理器写出时需要实例根目录")
        by_source: dict[str, list[TranslatedItem]] = {}
        for it in items:
            by_source.setdefault(it.entry.source, []).append(it)
        for rel, group in sorted(by_source.items()):
            obj = self._load(sink.root, rel)
            if obj is None:
                continue
            changed = False
            for it in group:
                if it.text == it.entry.original:
                    continue
                try:
                    _set(obj, json.loads(it.entry.locator), it.text)
                    changed = True
                except (KeyError, IndexError, ValueError, TypeError):
                    continue
            if changed:
                sink.add_instance_file(rel, (json.dumps(obj, ensure_ascii=False, indent=2) + "\n").encode("utf-8"))
```
